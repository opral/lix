# Run from the exact candidate repository root. Source scanner is independent
# of Lix compilation.
ORACLE=packages/lix/tests/forktree_stage2_execution_oracle
EVIDENCE=/tmp/forktree-stage2-execution-evidence
mkdir -p "$EVIDENCE"
rustc --edition=2024 -D warnings "$ORACLE/main.rs" -o "$EVIDENCE/residue-oracle"
"$EVIDENCE/residue-oracle" self-test
rustc --edition=2024 -D warnings "$ORACLE/compiler_dependency_oracle.rs" -o "$EVIDENCE/dependency-oracle"
"$EVIDENCE/dependency-oracle"
bash "$ORACLE/verify_landed_1258_map.sh" "$PWD"

# Exact pre-cut authority budgets. These outputs are frozen in the package and
# must be regenerated byte-for-byte from the pinned worktrees.
"$EVIDENCE/residue-oracle" budget /absolute/path/to/main-b5e > "$EVIDENCE/main-b5e-budget.tsv"
"$EVIDENCE/residue-oracle" definitions /absolute/path/to/main-b5e > "$EVIDENCE/main-b5e-deleted-module-definitions.tsv"
"$EVIDENCE/residue-oracle" budget /absolute/path/to/stage1-138b > "$EVIDENCE/stage1-138b-budget.tsv"
"$EVIDENCE/residue-oracle" definitions /absolute/path/to/stage1-138b > "$EVIDENCE/stage1-138b-deleted-module-definitions.tsv"
diff -u "$ORACLE/PRE_CUT_B5E_BUDGET.tsv" "$EVIDENCE/main-b5e-budget.tsv"
diff -u "$ORACLE/PRE_CUT_B5E_DELETED_MODULE_DEFINITIONS.tsv" "$EVIDENCE/main-b5e-deleted-module-definitions.tsv"
diff -u "$ORACLE/PRE_CUT_STAGE1_BUDGET.tsv" "$EVIDENCE/stage1-138b-budget.tsv"
diff -u "$ORACLE/PRE_CUT_STAGE1_DELETED_MODULE_DEFINITIONS.tsv" "$EVIDENCE/stage1-138b-deleted-module-definitions.tsv"

# The mutable H4 checkpoint is expected red. The immutable cursor head and
# post-cursor main must be green before Stage 2 starts.
"$EVIDENCE/residue-oracle" cursor-baseline /absolute/path/to/cursor-checkpoint
"$EVIDENCE/residue-oracle" cursor-audit /absolute/path/to/immutable-cursor-head

# Pinned inputs must be red; this records rather than accepts their residue.
"$EVIDENCE/residue-oracle" baseline /absolute/path/to/main-b5e
"$EVIDENCE/residue-oracle" baseline /absolute/path/to/stage1-138b

# The first runnable candidate must be green.
"$EVIDENCE/residue-oracle" audit "$PWD"

# Resolve the rlib from the exact candidate target directory. Both commands
# MUST fail to compile. A stale rlib is invalid evidence.
CARGO_TARGET_DIR=/absolute/path/to/exact-candidate-target CARGO_BUILD_JOBS=2 cargo build -p lix --lib
LIX_RLIB=$(find "$CARGO_TARGET_DIR/debug/deps" -maxdepth 1 -name 'liblix-*.rlib' -printf '%T@ %p\n' | sort -nr | head -1 | cut -d' ' -f2-)
test -n "$LIX_RLIB"
if rustc --edition=2024 -D warnings -L "dependency=$CARGO_TARGET_DIR/debug/deps" "$ORACLE/space_forge_rejection.rs" --extern "lix=$LIX_RLIB" -o "$EVIDENCE/space-forge" 2>"$EVIDENCE/space-forge.stderr"; then exit 1; fi
rg 'E0423|E0624|private|cannot initialize|not found' "$EVIDENCE/space-forge.stderr"
if rustc --edition=2024 -D warnings -L "dependency=$CARGO_TARGET_DIR/debug/deps" "$ORACLE/old_scan_compile_rejection.rs" --extern "lix=$LIX_RLIB" -o "$EVIDENCE/old-scan" 2>"$EVIDENCE/old-scan.stderr"; then exit 1; fi
rg 'E0432|E0599|unresolved import|no method named' "$EVIDENCE/old-scan.stderr"

# The sole positive cursor API and authenticated restart shape must compile.
rustc --edition=2024 -D warnings -L "dependency=$CARGO_TARGET_DIR/debug/deps" "$ORACLE/cursor_begin_scan_compile.rs" --extern "lix=$LIX_RLIB" -o "$EVIDENCE/cursor-begin-scan"
rustc --edition=2024 -D warnings -L "dependency=$CARGO_TARGET_DIR/debug/deps" "$ORACLE/cursor_restart_key_compile.rs" --extern "lix=$LIX_RLIB" -o "$EVIDENCE/cursor-restart-key"

# Caller-owned continuation state must remain unrepresentable.
if rustc --edition=2024 -D warnings -L "dependency=$CARGO_TARGET_DIR/debug/deps" "$ORACLE/cursor_resume_field_rejection.rs" --extern "lix=$LIX_RLIB" -o "$EVIDENCE/cursor-resume" 2>"$EVIDENCE/cursor-resume.stderr"; then exit 1; fi
rg 'E0560|has no field|unknown field' "$EVIDENCE/cursor-resume.stderr"

# Only after zero residue and both expected compile failures:
CARGO_TARGET_DIR=/absolute/path/to/exact-candidate-target CARGO_BUILD_JOBS=2 cargo check -p lix --all-features
CARGO_TARGET_DIR=/absolute/path/to/exact-candidate-target CARGO_BUILD_JOBS=2 cargo clippy -p lix --all-targets --all-features -- -D warnings
CARGO_TARGET_DIR=/absolute/path/to/exact-candidate-target CARGO_BUILD_JOBS=2 cargo test -p lix forktree::tests --lib -j2 -- --test-threads=1
CARGO_TARGET_DIR=/absolute/path/to/exact-candidate-target CARGO_BUILD_JOBS=2 cargo test -p lix --test integration sealed_owner_violations_are_empty -- --nocapture
CARGO_TARGET_DIR=/absolute/path/to/exact-candidate-target CARGO_BUILD_JOBS=2 cargo test -p lix_tests --test forktree_stage1_application_oracle --no-run
CARGO_TARGET_DIR=/absolute/path/to/exact-candidate-target cargo test -p lix_tests --test forktree_stage1_application_oracle forktree_stage1_application_rocksdb -- --exact --nocapture --test-threads=1
CARGO_TARGET_DIR=/absolute/path/to/exact-candidate-target cargo test -p lix_tests --test forktree_stage1_application_oracle forktree_stage1_application_slatedb -- --exact --nocapture --test-threads=1
cargo fmt --all -- --check
git diff --check
