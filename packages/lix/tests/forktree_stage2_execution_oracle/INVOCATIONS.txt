# Run from an exact candidate repository root. This package is test/static only.
ORACLE=packages/lix/tests/forktree_stage2_execution_oracle
EVIDENCE=/tmp/forktree-stage2-execution-evidence
mkdir -p "$EVIDENCE"

rustc --edition=2024 -D warnings "$ORACLE/main.rs" -o "$EVIDENCE/residue-oracle"
"$EVIDENCE/residue-oracle" self-test
rustc --edition=2024 -D warnings "$ORACLE/compiler_dependency_oracle.rs" -o "$EVIDENCE/dependency-oracle"
"$EVIDENCE/dependency-oracle"
bash "$ORACLE/verify_landed_1258_map.sh" "$PWD"

# Regenerate every pinned pre-cut calibration. All four are intentionally red.
for SPEC in \
  b5:/absolute/path/to/main-b5e \
  stage1:/absolute/path/to/stage1-138b \
  e871:/absolute/path/to/main-e871 \
  main803d:/absolute/path/to/main-803d
do
  NAME=${SPEC%%:*}
  ROOT=${SPEC#*:}
  timeout 20m "$EVIDENCE/residue-oracle" budget "$ROOT" > "$EVIDENCE/$NAME-budget.tsv"
  timeout 20m "$EVIDENCE/residue-oracle" definitions "$ROOT" > "$EVIDENCE/$NAME-definitions.tsv"
  timeout 20m "$EVIDENCE/residue-oracle" baseline "$ROOT" > "$EVIDENCE/$NAME-baseline.txt"
done

diff -u "$ORACLE/PRE_CUT_B5E_BUDGET.tsv" "$EVIDENCE/b5-budget.tsv"
diff -u "$ORACLE/PRE_CUT_B5E_DELETED_MODULE_DEFINITIONS.tsv" "$EVIDENCE/b5-definitions.tsv"
diff -u "$ORACLE/PRE_CUT_STAGE1_BUDGET.tsv" "$EVIDENCE/stage1-budget.tsv"
diff -u "$ORACLE/PRE_CUT_STAGE1_DELETED_MODULE_DEFINITIONS.tsv" "$EVIDENCE/stage1-definitions.tsv"
diff -u "$ORACLE/PRE_CUT_E871_BUDGET.tsv" "$EVIDENCE/e871-budget.tsv"
diff -u "$ORACLE/PRE_CUT_E871_DELETED_MODULE_DEFINITIONS.tsv" "$EVIDENCE/e871-definitions.tsv"
diff -u "$ORACLE/PRE_CUT_MAIN803D_BUDGET.tsv" "$EVIDENCE/main803d-budget.tsv"
diff -u "$ORACLE/PRE_CUT_MAIN803D_DELETED_MODULE_DEFINITIONS.tsv" "$EVIDENCE/main803d-definitions.tsv"
diff -u "$ORACLE/PRE_CUT_B5E_BASELINE.txt" "$EVIDENCE/b5-baseline.txt"
diff -u "$ORACLE/PRE_CUT_STAGE1_BASELINE.txt" "$EVIDENCE/stage1-baseline.txt"
diff -u "$ORACLE/PRE_CUT_E871_BASELINE.txt" "$EVIDENCE/e871-baseline.txt"
diff -u "$ORACLE/PRE_CUT_MAIN803D_BASELINE.txt" "$EVIDENCE/main803d-baseline.txt"

# All four pre-cut trees intentionally fail the same typed CLI route contract.
for ROOT in \
  /absolute/path/to/main-b5e \
  /absolute/path/to/stage1-138b \
  /absolute/path/to/main-e871 \
  /absolute/path/to/main-803d
do
  if "$EVIDENCE/residue-oracle" cli-routing-audit "$ROOT" > "$EVIDENCE/cli-routing.txt" 2>&1; then exit 1; fi
  diff -u "$ORACLE/PRE_CUT_CLI_ROUTING_BASELINE.txt" "$EVIDENCE/cli-routing.txt"
done

# The immutable cursor successor must be green before Stage 2 starts.
"$EVIDENCE/residue-oracle" cursor-audit /absolute/path/to/immutable-cursor-head

# The first runnable Stage-2 candidate must be green for all residue and for
# executable public-facade ownership.
"$EVIDENCE/residue-oracle" audit "$PWD"
"$EVIDENCE/residue-oracle" semantic-audit "$PWD"
"$EVIDENCE/residue-oracle" cli-routing-audit "$PWD"

# The item-scoped source audit above proves the call graph. This typed runtime
# gate independently proves concrete RocksDB ownership for normal CLI init/open
# and durable cold reopen for both RocksDB and explicitly selected SlateDB.
# It cannot compile while open_lix_at still returns FileLix/FileStorage.
CARGO_TARGET_DIR=/absolute/path/to/exact-candidate-target CARGO_BUILD_JOBS=2 cargo test -p lix_cli --test forktree_stage2_cli_storage_routing --no-run
CARGO_TARGET_DIR=/absolute/path/to/exact-candidate-target cargo test -p lix_cli --test forktree_stage2_cli_storage_routing normal_cli_init_open_is_rocksdb_and_cold_reopens -- --exact --nocapture --test-threads=1
CARGO_TARGET_DIR=/absolute/path/to/exact-candidate-target cargo test -p lix_cli --test forktree_stage2_cli_storage_routing explicit_slatedb_selection_cold_reopens -- --exact --nocapture --test-threads=1
CARGO_TARGET_DIR=/absolute/path/to/exact-candidate-target CARGO_BUILD_JOBS=2 cargo clippy -p lix_cli --test forktree_stage2_cli_storage_routing -- -D warnings

# Build the exact candidate before compile probes; stale rlibs are invalid.
CARGO_TARGET_DIR=/absolute/path/to/exact-candidate-target CARGO_BUILD_JOBS=2 cargo build -p lix --lib
LIX_RLIB=$(find "$CARGO_TARGET_DIR/debug/deps" -maxdepth 1 -name 'liblix-*.rlib' -printf '%T@ %p\n' | sort -nr | head -1 | cut -d' ' -f2-)
test -n "$LIX_RLIB"

# Both superseded API shapes must be uncallable.
if rustc --edition=2024 -D warnings -L "dependency=$CARGO_TARGET_DIR/debug/deps" "$ORACLE/space_forge_rejection.rs" --extern "lix=$LIX_RLIB" -o "$EVIDENCE/space-forge" 2>"$EVIDENCE/space-forge.stderr"; then exit 1; fi
rg 'E0423|E0624|private|cannot initialize|not found' "$EVIDENCE/space-forge.stderr"
if rustc --edition=2024 -D warnings -L "dependency=$CARGO_TARGET_DIR/debug/deps" "$ORACLE/old_scan_compile_rejection.rs" --extern "lix=$LIX_RLIB" -o "$EVIDENCE/old-scan" 2>"$EVIDENCE/old-scan.stderr"; then exit 1; fi
rg 'E0432|E0599|unresolved import|no method named' "$EVIDENCE/old-scan.stderr"

# The one streaming cursor and authenticated exclusive restart compile.
rustc --edition=2024 -D warnings -L "dependency=$CARGO_TARGET_DIR/debug/deps" "$ORACLE/cursor_begin_scan_compile.rs" --extern "lix=$LIX_RLIB" -o "$EVIDENCE/cursor-begin-scan"
rustc --edition=2024 -D warnings -L "dependency=$CARGO_TARGET_DIR/debug/deps" "$ORACLE/cursor_restart_key_compile.rs" --extern "lix=$LIX_RLIB" -o "$EVIDENCE/cursor-restart-key"

# Caller-owned/durable continuation state is unrepresentable.
if rustc --edition=2024 -D warnings -L "dependency=$CARGO_TARGET_DIR/debug/deps" "$ORACLE/cursor_resume_field_rejection.rs" --extern "lix=$LIX_RLIB" -o "$EVIDENCE/cursor-resume" 2>"$EVIDENCE/cursor-resume.stderr"; then exit 1; fi
rg 'E0432|E0560|has no field|unknown field' "$EVIDENCE/cursor-resume.stderr"

# Final retained-workspace boundary. SQLite and FileStorage are deletion
# requirements, not qualified adapters or compatibility paths.
test ! -e packages/sqlite-storage
! rg -n 'lix_storage_sqlite|StorageProfile::SQLite|LixStorageProfile::SQLite|SQLiteTempBenchStorage|ChangelogScoreStorage::SQLite|FileLix|FileStorage(Read|Write)?' packages Cargo.toml
if cargo metadata --no-deps --format-version 1 | rg -q 'sqlite-storage|lix_storage_sqlite'; then exit 1; fi
"$EVIDENCE/residue-oracle" cli-routing-audit "$PWD"

CARGO_TARGET_DIR=/absolute/path/to/exact-candidate-target CARGO_BUILD_JOBS=2 cargo clippy --workspace --all-targets --all-features -- -D warnings
CARGO_TARGET_DIR=/absolute/path/to/exact-candidate-target CARGO_BUILD_JOBS=2 cargo test -p lix forktree::tests --lib -j2 -- --test-threads=1
CARGO_TARGET_DIR=/absolute/path/to/exact-candidate-target CARGO_BUILD_JOBS=2 cargo test -p lix --test integration sealed_owner_violations_are_empty -- --nocapture
